cmake_minimum_required(VERSION 3.25)
project(pcut VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
option(PCUT_BUILD_EXAMPLES "Build the model example executables" ON)
option(PCUT_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(Boost 1.81 REQUIRED CONFIG)
find_package(Threads REQUIRED)
add_library(pcut src/coefficients.cpp src/model.cpp src/effective.cpp src/coefficient_program.cpp src/lattice.cpp src/white_graph.cpp src/polynomial.cpp src/linked.cpp src/models.cpp src/sectors.cpp src/operator.cpp src/hubbard.cpp)
add_library(pcut::pcut ALIAS pcut)
target_compile_features(pcut PUBLIC cxx_std_20)
target_include_directories(pcut PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(pcut PUBLIC Eigen3::Eigen Boost::headers Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(pcut PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(PCUT_SANITIZE)
    target_compile_options(pcut PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pcut PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(PCUT_BUILD_EXAMPLES)
  add_executable(pcut_chain examples/chain.cpp)
  target_link_libraries(pcut_chain PRIVATE pcut::pcut)
  add_executable(pcut_white_sweep examples/white_sweep.cpp)
  target_link_libraries(pcut_white_sweep PRIVATE pcut::pcut)
  add_executable(pcut_hubbard examples/hubbard.cpp)
  target_link_libraries(pcut_hubbard PRIVATE pcut::pcut)
endif()
if(BUILD_TESTING)
  find_package(Catch2 3 QUIET)
  if(NOT Catch2_FOUND)
    include(FetchContent)
    FetchContent_Declare(Catch2
      URL https://codeload.github.com/catchorg/Catch2/tar.gz/2b60af89e23d28eefc081bc930831ee9d45ea58b
      URL_HASH SHA256=553e6c94498c42edaecc9c8459f1a320a7e4cf49155635cf7123212bb9f09cf2)
    FetchContent_MakeAvailable(Catch2)
  endif()
  add_executable(pcut_tests tests/coefficients.cpp tests/model.cpp tests/linked.cpp tests/white_graph.cpp tests/chain.cpp tests/sectors.cpp tests/hubbard.cpp)
  target_link_libraries(pcut_tests PRIVATE pcut::pcut Catch2::Catch2WithMain)
  target_compile_definitions(pcut_tests PRIVATE PCUT_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  include(Catch)
  catch_discover_tests(pcut_tests)
endif()
install(TARGETS pcut EXPORT pcutTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT pcutTargets NAMESPACE pcut:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pcut)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/pcutConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/pcutConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pcut)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/pcutConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pcutConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/pcutConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pcut)
