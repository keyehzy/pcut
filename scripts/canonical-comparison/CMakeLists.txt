cmake_minimum_required(VERSION 3.25)
project(canonical_comparison LANGUAGES C CXX)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(Boost 1.81 REQUIRED CONFIG)
find_package(Threads REQUIRED)
foreach(engine IN ITEMS previous current)
  foreach(kind IN ITEMS isolated topology sweep)
    if(kind STREQUAL isolated)
      set(source measure.cpp)
    elseif(kind STREQUAL topology)
      set(source topology.cpp)
    else()
      set(source ../engine-comparison/measure.cpp)
    endif()
    add_executable(${engine}_${kind} ${source})
    target_compile_features(${engine}_${kind} PRIVATE cxx_std_20)
    target_include_directories(${engine}_${kind} PRIVATE ${${engine}_ROOT}/include)
    target_link_libraries(${engine}_${kind} PRIVATE ${${engine}_ROOT}/build/release/libpcut.a Eigen3::Eigen Boost::headers Threads::Threads)
    if(engine STREQUAL current)
      target_link_libraries(${engine}_${kind} PRIVATE ${current_ROOT}/build/release/libpcut_nauty.a)
    endif()
  endforeach()
endforeach()
# The backend microbenchmark uses the exact same prepared colored incidence
# graphs. Traces remains tooling-only, with no production selection switch.
add_library(traces STATIC ${nauty_SOURCE}/traces.c ${nauty_SOURCE}/gtools.c)
target_include_directories(traces PRIVATE ${nauty_BINARY} ${nauty_SOURCE})
target_compile_options(traces PRIVATE -include ${nauty_BINARY}/gtools.h)
add_executable(backends backends.cpp)
target_compile_features(backends PRIVATE cxx_std_20)
target_include_directories(backends PRIVATE ${current_ROOT}/include ${current_ROOT}/src ${nauty_BINARY} ${nauty_SOURCE})
target_link_libraries(backends PRIVATE traces ${current_ROOT}/build/release/libpcut.a ${current_ROOT}/build/release/libpcut_nauty.a Eigen3::Eigen Boost::headers Threads::Threads)
