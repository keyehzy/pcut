cmake_minimum_required(VERSION 3.25)
project(engine_comparison LANGUAGES CXX)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(Boost 1.81 REQUIRED CONFIG)
find_package(Threads REQUIRED)
foreach(engine IN ITEMS previous current)
  add_executable(${engine} measure.cpp)
  target_compile_features(${engine} PRIVATE cxx_std_20)
  target_link_libraries(${engine} PRIVATE Eigen3::Eigen Boost::headers Threads::Threads)
  target_include_directories(${engine} PRIVATE ${${engine}_ROOT}/include)
  target_link_libraries(${engine} PRIVATE ${${engine}_ROOT}/build/release/libpcut.a)
endforeach()
target_compile_definitions(previous PRIVATE PREVIOUS)
